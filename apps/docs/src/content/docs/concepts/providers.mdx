## Available providers

### OpenAI

```bash
pnpm add @intentform/provider-openai
```

```typescript

  apiKey: process.env.OPENAI_API_KEY!,
  model: 'gpt-4o-mini', // default
})
```

### Ollama (local)

```bash
pnpm add @intentform/provider-ollama
```

```typescript

ollamaProvider({
  model: 'llama3.2',
  baseUrl: 'http://localhost:11434', // default
})
```

:::note[Ollama]
Ollama runs locally — no API key required. Model quality varies; use a tier fallback for production. See [Confidence Tiers](/concepts/confidence-tiers).
:::

## Custom provider

Implement the `AiProvider` interface to integrate any AI backend:

```typescript
import type { AiProvider, GenerateInput } from '@intentform/core'

const myProvider: AiProvider = {
  async generateStructured(input: GenerateInput) {
    const response = await fetch('/api/intent', {
      method: 'POST',
      body: JSON.stringify(input),
    })
    const json = await response.json()
    return {
      data: json.data,
      confidence: json.confidence,
      usage: {
        tokensIn: json.usage?.input_tokens ?? 0,
        tokensOut: json.usage?.output_tokens ?? 0,
      },
    }
  },
}
```

This pattern is useful for keeping API keys server-side. See [Keep API Key Server-Side](/recipes/server-side-key).

## Provider interface

```typescript
interface AiProvider {
  generateStructured(input: GenerateInput): Promise<{
    data: unknown
    confidence: number
    usage?: { tokensIn: number; tokensOut: number; cost?: number }
  }>
}
```
